# SafeChain
#
# Almost all of this product sits behind a sign-in, so a crawler cannot reach it whatever
# this file says. The rules below exist for the handful of pages that are genuinely public,
# and to keep two kinds of URL out of any index that might otherwise pick them up.
#
# Nothing here is a security control. A path listed as Disallow is a request, not a barrier,
# and every route name is already in the JavaScript bundle any visitor can download. The
# actual protection is the API's authorization, not this file.

User-agent: *

# Single-use credentials live in these URLs. An invitation link creates an account and a
# reset link takes one over, so neither should ever be crawled, cached or indexed.
Disallow: /accept-invitation/
Disallow: /reset-password

# Signed-in surface. A crawler gets a sign-in redirect from all of it; saying so saves the
# request and keeps app paths out of search results.
Disallow: /incidents
Disallow: /actions
Disallow: /assets
Disallow: /audits
Disallow: /training
Disallow: /permits
Disallow: /reports
Disallow: /visitors
Disallow: /contractors
Disallow: /employees
Disallow: /admin
Disallow: /organization
Disallow: /notifications
Disallow: /account
Disallow: /platform
Disallow: /near-miss
Disallow: /design

# Public and intended to be readable: the sign-in page, the PDPA notice and the terms.
Allow: /$
Allow: /login
Allow: /privacy
Allow: /terms

# The sitemap is at /sitemap.xml. There is no Sitemap: line here because the directive
# requires an absolute URL and this file is a static asset with no way to learn the
# deployment's hostname — the sitemap itself is generated by nginx from APP_PUBLIC_URL for
# exactly that reason. Crawlers check /sitemap.xml by convention regardless.
